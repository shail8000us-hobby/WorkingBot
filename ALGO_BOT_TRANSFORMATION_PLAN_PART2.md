# bot/strategy/options_strategy_interface.py

from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum
from typing import Optional, List, Dict, Any
import pandas as pd

class OptionType(Enum):
    CALL = "call"
    PUT = "put"

class SignalAction(Enum):
    SELL_PUT = "sell_put"
    BUY_PUT = "buy_put"
    SELL_CALL = "sell_call"
    BUY_CALL = "buy_call"
    CLOSE = "close"
    ROLL = "roll"
    HOLD = "hold"

@dataclass
class OptionSignal:
    """Signal generated by strategy"""
    action: SignalAction
    underlying: str
    strike: Optional[float] = None
    expiry: Optional[str] = None
    delta_target: Optional[float] = None
    dte_target: Optional[int] = None
    size: int = 1
    confidence: float = 0.5
    reason: str = ""
    urgency: str = "normal"  # low, normal, high, critical

@dataclass
class MarketSnapshot:
    """Current market state for strategy evaluation"""
    underlying: str
    spot_price: float
    iv_current: float
    iv_percentile_30d: float
    iv_percentile_60d: float
    iv_percentile_90d: float
    term_structure: str  # "contango", "backwardation", "flat"
    realized_vol_20d: float
    vix_equivalent: Optional[float] = None
    trend: str = "neutral"  # bearish, neutral, bullish
    timestamp: Optional[str] = None

@dataclass
class PositionSnapshot:
    """Current position state"""
    symbol: str
    underlying: str
    option_type: OptionType
    strike: float
    expiry: str
    dte: int
    size: int  # positive = long, negative = short
    entry_price: float
    current_price: float
    unrealized_pnl: float
    pnl_percent: float
    delta: float
    gamma: float
    theta: float
    vega: float
    iv: float

class IOptionsStrategy(ABC):
    """
    Base interface for options trading strategies.
    
    Implement this to create your own strategies.
    """
    
    # === CONFIGURATION ===
    name: str = "unnamed_strategy"
    version: str = "1.0.0"
    description: str = ""
    
    # Risk parameters
    max_position_size: int = 10
    max_portfolio_delta: float = 100.0
    max_single_trade_loss: float = 500.0
    max_daily_loss: float = 2000.0
    
    # Entry parameters
    min_iv_percentile: float = 50.0
    max_iv_percentile: float = 100.0
    min_dte: int = 7
    max_dte: int = 60
    target_delta_range: tuple = (-0.35, -0.15)
    
    # Exit parameters
    take_profit_percent: float = 50.0
    stop_loss_percent: float = 200.0
    close_at_dte: int = 7
    
    def __init__(self, config: Dict[str, Any] = None):
        """Initialize strategy with optional config override"""
        if config:
            for key, value in config.items():
                if hasattr(self, key):
                    setattr(self, key, value)
    
    # === ABSTRACT METHODS (Must Implement) ===
    
    @abstractmethod
    def scan_for_entries(
        self, 
        market: MarketSnapshot,
        chain: pd.DataFrame,  # Options chain DataFrame
        current_positions: List[PositionSnapshot]
    ) -> List[OptionSignal]:
        """
        Scan options chain for entry opportunities.
        
        Args:
            market: Current market conditions
            chain: Full options chain with columns:
                   [symbol, strike, expiry, type, bid, ask, iv, delta, gamma, theta, vega]
            current_positions: List of current positions
        
        Returns:
            List of entry signals (can be empty)
        """
        pass
    
    @abstractmethod
    def evaluate_position(
        self,
        position: PositionSnapshot,
        market: MarketSnapshot
    ) -> OptionSignal:
        """
        Evaluate an existing position for exit/adjustment.
        
        Args:
            position: Current position state
            market: Current market conditions
        
        Returns:
            Signal for this position (HOLD, CLOSE, ROLL, etc.)
        """
        pass
    
    # === OPTIONAL METHODS (Override if needed) ===
    
    def calculate_position_size(
        self,
        signal: OptionSignal,
        market: MarketSnapshot,
        account_balance: float
    ) -> int:
        """Calculate appropriate position size"""
        # Default: fixed size
        return min(signal.size, self.max_position_size)
    
    def should_trade_now(self, market: MarketSnapshot) -> bool:
        """Check if market conditions allow trading"""
        # Default: always allow
        return True
    
    def get_preferred_strikes(
        self,
        market: MarketSnapshot,
        expiry: str
    ) -> List[float]:
        """Get list of preferred strikes for given expiry"""
        # Default: return strikes matching target delta
        return []
    
    def on_fill(self, order_result: Dict[str, Any]) -> None:
        """Called when an order is filled"""
        pass
    
    def on_error(self, error: Exception) -> None:
        """Called when an error occurs"""
        pass

# bot/strategy/theta_harvester.py

from .options_strategy_interface import (
    IOptionsStrategy, OptionSignal, SignalAction,
    MarketSnapshot, PositionSnapshot, OptionType
)
import pandas as pd
from typing import List

class ThetaHarvesterStrategy(IOptionsStrategy):
    """
    Theta Harvesting Strategy
    
    Sells OTM puts when IV is elevated, captures theta decay.
    
    Entry Conditions:
    - IV percentile >= 70%
    - Delta between -0.30 and -0.15
    - DTE between 21 and 45 days
    - Underlying not in strong downtrend
    
    Exit Conditions:
    - Take profit at 50% of premium received
    - Stop loss at 200% of premium (2x loss)
    - Close at 7 DTE regardless of P&L
    - Roll if delta exceeds -0.50
    """
    
    name = "theta_harvester"
    version = "1.0.0"
    description = "Sell OTM puts in high IV environments"
    
    # Strategy-specific parameters
    min_iv_percentile = 70.0
    target_delta_range = (-0.30, -0.15)
    min_dte = 21
    max_dte = 45
    take_profit_percent = 50.0
    stop_loss_percent = 200.0
    close_at_dte = 7
    roll_at_delta = -0.50
    
    def scan_for_entries(
        self,
        market: MarketSnapshot,
        chain: pd.DataFrame,
        current_positions: List[PositionSnapshot]
    ) -> List[OptionSignal]:
        """Scan for put selling opportunities"""
        
        signals = []
        
        # Check IV condition
        if market.iv_percentile_30d < self.min_iv_percentile:
            return signals  # No entries in low IV
        
        # Check trend (don't sell puts in crash)
        if market.trend == "bearish":
            return signals
        
        # Filter chain for puts only
        puts = chain[chain['type'] == 'put'].copy()
        
        # Filter by DTE
        puts = puts[
            (puts['dte'] >= self.min_dte) & 
            (puts['dte'] <= self.max_dte)
        ]
        
        # Filter by delta
        puts = puts[
            (puts['delta'] >= self.target_delta_range[0]) &
            (puts['delta'] <= self.target_delta_range[1])
        ]
        
        if puts.empty:
            return signals
        
        # Check if we already have position in this underlying
        existing = [p for p in current_positions if p.underlying == market.underlying]
        if len(existing) >= 2:  # Max 2 positions per underlying
            return signals
        
        # Select best candidate (highest premium at target delta)
        puts = puts.sort_values('bid', ascending=False)
        best = puts.iloc[0]
        
        # Generate signal
        signal = OptionSignal(
            action=SignalAction.SELL_PUT,
            underlying=market.underlying,
            strike=best['strike'],
            expiry=best['expiry'],
            delta_target=best['delta'],
            dte_target=int(best['dte']),
            size=1,  # Will be sized by position sizer
            confidence=min(1.0, market.iv_percentile_30d / 100),
            reason=f"IV at {market.iv_percentile_30d:.0f}th pctl, delta {best['delta']:.2f}, {int(best['dte'])} DTE",
            urgency="normal"
        )
        
        signals.append(signal)
        return signals
    
    def evaluate_position(
        self,
        position: PositionSnapshot,
        market: MarketSnapshot
    ) -> OptionSignal:
        """Evaluate existing position for exit or adjustment"""
        
        # Take Profit
        if position.pnl_percent >= self.take_profit_percent:
            return OptionSignal(
                action=SignalAction.CLOSE,
                underlying=position.underlying,
                strike=position.strike,
                expiry=position.expiry,
                confidence=1.0,
                reason=f"Take profit: {position.pnl_percent:.1f}% >= {self.take_profit_percent}%",
                urgency="high"
            )
        
        # Stop Loss
        if position.pnl_percent <= -self.stop_loss_percent:
            return OptionSignal(
                action=SignalAction.CLOSE,
                underlying=position.underlying,
                strike=position.strike,
                expiry=position.expiry,
                confidence=1.0,
                reason=f"Stop loss: {position.pnl_percent:.1f}% <= -{self.stop_loss_percent}%",
                urgency="critical"
            )
        
        # Time Exit
        if position.dte <= self.close_at_dte:
            return OptionSignal(
                action=SignalAction.CLOSE,
                underlying=position.underlying,
                strike=position.strike,
                expiry=position.expiry,
                confidence=0.9,
                reason=f"Time exit: {position.dte} DTE <= {self.close_at_dte}",
                urgency="normal"
            )
        
        # Roll if delta too high (position getting risky)
        if position.delta <= self.roll_at_delta:
            return OptionSignal(
                action=SignalAction.ROLL,
                underlying=position.underlying,
                strike=position.strike,
                expiry=position.expiry,
                delta_target=-0.20,  # Roll to safer delta
                confidence=0.8,
                reason=f"Delta {position.delta:.2f} exceeded threshold {self.roll_at_delta}",
                urgency="high"
            )
        
        # No action needed
        return OptionSignal(
            action=SignalAction.HOLD,
            underlying=position.underlying,
            strike=position.strike,
            expiry=position.expiry,
            confidence=0.5,
            reason=f"Position healthy: {position.pnl_percent:.1f}% P&L, {position.dte} DTE",
            urgency="low"
        )


# Strategy factory
STRATEGIES = {
    "theta_harvester": ThetaHarvesterStrategy,
    # Add more strategies here
}

def get_strategy(name: str, config: dict = None) -> IOptionsStrategy:
    """Factory function to create strategy instances"""
    if name not in STRATEGIES:
        raise ValueError(f"Unknown strategy: {name}")
    return STRATEGIES[name](config)

# bot/backtest/engine.py

from dataclasses import dataclass
from typing import List, Dict, Optional
import pandas as pd
import numpy as np
from datetime import datetime, timedelta

@dataclass
class BacktestConfig:
    """Backtest configuration"""
    strategy_name: str
    start_date: datetime
    end_date: datetime
    initial_capital: float
    commission_rate: float = 0.0005  # 0.05%
    slippage_percent: float = 0.1    # 0.1% slippage
    underlying: str = "BTC"

@dataclass
class BacktestTrade:
    """Single trade in backtest"""
    entry_time: datetime
    exit_time: Optional[datetime]
    symbol: str
    side: str  # 'buy' or 'sell'
    size: int
    entry_price: float
    exit_price: Optional[float]
    pnl: Optional[float]
    pnl_percent: Optional[float]
    exit_reason: Optional[str]

@dataclass
class BacktestResult:
    """Complete backtest results"""
    config: BacktestConfig
    trades: List[BacktestTrade]
    
    # Performance metrics
    total_pnl: float
    total_return_percent: float
    sharpe_ratio: float
    sortino_ratio: float
    max_drawdown_percent: float
    max_drawdown_duration_days: int
    win_rate: float
    profit_factor: float
    avg_trade_pnl: float
    avg_winner: float
    avg_loser: float
    largest_winner: float
    largest_loser: float
    
    # Time series
    equity_curve: pd.Series
    drawdown_curve: pd.Series
    monthly_returns: pd.Series

class BacktestEngine:
    """
    Event-driven backtesting engine for options strategies.
    """
    
    def __init__(self, config: BacktestConfig):
        self.config = config
        self.capital = config.initial_capital
        self.positions: List[Dict] = []
        self.trades: List[BacktestTrade] = []
        self.equity_history: List[tuple] = []
    
    def load_data(self) -> pd.DataFrame:
        """Load historical data for backtest period"""
        # Implementation: Query database for historical options data
        pass
    
    def run(self, strategy) -> BacktestResult:
        """Run backtest with given strategy"""
        
        data = self.load_data()
        
        # Sort by timestamp
        data = data.sort_values('timestamp')
        
        # Group by day for simulation
        for date, day_data in data.groupby(data['timestamp'].dt.date):
            # Get market snapshot for this day
            market = self._build_market_snapshot(day_data)
            
            # Get options chain for this day
            chain = self._build_options_chain(day_data)
            
            # Current positions
            current_positions = self._build_position_snapshots()
            
            # 1. Evaluate existing positions
            for pos in list(self.positions):
                pos_snapshot = self._position_to_snapshot(pos, day_data)
                signal = strategy.evaluate_position(pos_snapshot, market)
                
                if signal.action in [SignalAction.CLOSE, SignalAction.ROLL]:
                    self._close_position(pos, day_data, signal.reason)
                    
                    if signal.action == SignalAction.ROLL:
                        # Open new position at target delta
                        self._open_roll_position(pos, signal, chain)
            
            # 2. Scan for new entries
            signals = strategy.scan_for_entries(market, chain, current_positions)
            
            for signal in signals:
                if self._can_open_position(signal):
                    self._open_position(signal, chain)
            
            # 3. Record equity
            equity = self._calculate_equity(day_data)
            self.equity_history.append((date, equity))
        
        # Close remaining positions at end
        for pos in list(self.positions):
            self._close_position(pos, data.iloc[-1], "backtest_end")
        
        # Calculate metrics
        return self._calculate_results()
    
    def _calculate_results(self) -> BacktestResult:
        """Calculate performance metrics from trades"""
        
        if not self.trades:
            return BacktestResult(
                config=self.config,
                trades=[],
                total_pnl=0, total_return_percent=0,
                sharpe_ratio=0, sortino_ratio=0,
                max_drawdown_percent=0, max_drawdown_duration_days=0,
                win_rate=0, profit_factor=0,
                avg_trade_pnl=0, avg_winner=0, avg_loser=0,
                largest_winner=0, largest_loser=0,
                equity_curve=pd.Series(), drawdown_curve=pd.Series(),
                monthly_returns=pd.Series()
            )
        
        # Build equity curve
        equity_df = pd.DataFrame(self.equity_history, columns=['date', 'equity'])
        equity_df.set_index('date', inplace=True)
        equity_curve = equity_df['equity']
        
        # Calculate returns
        returns = equity_curve.pct_change().dropna()
        
        # Metrics
        total_pnl = equity_curve.iloc[-1] - self.config.initial_capital
        total_return = (equity_curve.iloc[-1] / self.config.initial_capital - 1) * 100
        
        # Sharpe (assuming 0% risk-free rate)
        sharpe = np.sqrt(252) * returns.mean() / returns.std() if returns.std() > 0 else 0
        
        # Sortino (downside deviation)
        downside = returns[returns < 0]
        sortino = np.sqrt(252) * returns.mean() / downside.std() if len(downside) > 0 and downside.std() > 0 else 0
        
        # Drawdown
        running_max = equity_curve.cummax()
        drawdown = (equity_curve - running_max) / running_max * 100
        max_dd = drawdown.min()
        
        # Win rate
        pnls = [t.pnl for t in self.trades if t.pnl is not None]
        winners = [p for p in pnls if p > 0]
        losers = [p for p in pnls if p < 0]
        win_rate = len(winners) / len(pnls) * 100 if pnls else 0
        
        # Profit factor
        gross_profit = sum(winners) if winners else 0
        gross_loss = abs(sum(losers)) if losers else 1
        profit_factor = gross_profit / gross_loss if gross_loss > 0 else 999
        
        return BacktestResult(
            config=self.config,
            trades=self.trades,
            total_pnl=total_pnl,
            total_return_percent=total_return,
            sharpe_ratio=sharpe,
            sortino_ratio=sortino,
            max_drawdown_percent=max_dd,
            max_drawdown_duration_days=0,  # TODO: calculate
            win_rate=win_rate,
            profit_factor=profit_factor,
            avg_trade_pnl=np.mean(pnls) if pnls else 0,
            avg_winner=np.mean(winners) if winners else 0,
            avg_loser=np.mean(losers) if losers else 0,
            largest_winner=max(winners) if winners else 0,
            largest_loser=min(losers) if losers else 0,
            equity_curve=equity_curve,
            drawdown_curve=drawdown,
            monthly_returns=returns.resample('M').sum() * 100
        )
