```python
"""
Bot Control Routes Blueprint

This module handles all API routes related to bot lifecycle management.

Routes:
- POST /api/bot/start - Start the trading bot
- POST /api/bot/stop - Stop the trading bot
- POST /api/bot/restart - Restart the trading bot
- GET  /api/bot/status - Get bot running status
- GET  /api/bots/status - Get status of all bots (bot + guardian + monitor)
- POST /api/bots/stop - Emergency stop all bots

Dependencies:
- Subprocess management for bot process control
- PID file reading for status checks
- Tmux session management
- File system access for logs and state files

Author: Refactored from app.py God file (8,850 lines)
Date: 2025-10-31
"""

import os
import sys
import logging
import subprocess
import signal
from pathlib import Path
from typing import Dict, Optional, List

from flask import Blueprint, jsonify, request

# Initialize logger
log = logging.getLogger(__name__)

# Create blueprint
bot_control_bp = Blueprint('bot_control', __name__)

# ============================================================================
# Constants
# ============================================================================

BOT_PID_FILE = Path("reports/bot.pid")
GUARDIAN_PID_FILE = Path("reports/guardian.pid")
MONITOR_PID_FILE = Path("reports/monitor.pid")

# ============================================================================
# Helper Functions (private to this blueprint)
# ============================================================================

def _is_process_running(pid: int) -> bool:
    """Check if process with given PID is running"""
    try:
        os.kill(pid, 0)
        return True
    except (OSError, ProcessLookupError):
        return False

def _read_pid_file(filepath: Path) -> Optional[int]:
    """Read PID from file, return None if file doesn't exist or invalid"""
    try:
        if filepath.exists():
            with open(filepath, 'r') as f:
                return int(f.read().strip())
    except (ValueError, IOError) as e:
        log.warning(f"Failed to read PID from {filepath}: {e}")
    return None

def _stop_process_by_pid(pid: int, timeout: int = 5) -> bool:
    """
    Stop process gracefully (SIGTERM), then forcefully (SIGKILL) if needed
    
    Returns:
        True if process stopped, False otherwise
    """
    try:
        # Try graceful shutdown first
        os.kill(pid, signal.SIGTERM)
        
        # Wait for process to exit
        import time
        for _ in range(timeout * 10):
            if not _is_process_running(pid):
                return True
            time.sleep(0.1)
        
        # Force kill if still running
        log.warning(f"Process {pid} didn't stop gracefully, force killing")
        os.kill(pid, signal.SIGKILL)
        return True
        
    except ProcessLookupError:
        return True  # Already stopped
    except Exception as e:
        log.error(f"Failed to stop process {pid}: {e}")
        return False

# ============================================================================
# Route Handlers
# ============================================================================

@bot_control_bp.route('/api/bot/status', methods=['GET'])
def get_bot_status():
    """
    Get current status of trading bot
    
    Returns:
        JSON response with bot running status and PID
    
    Example:
        GET /api/bot/status
        Response: {
            "running": true,
            "pid": 12345,
            "uptime": "2h 15m"
        }
    """
    try:
        pid = _read_pid_file(BOT_PID_FILE)
        
        if pid and _is_process_running(pid):
            # Calculate uptime from PID file modification time
            mtime = BOT_PID_FILE.stat().st_mtime
            uptime_seconds = int(time.time() - mtime)
            
            return jsonify({
                'running': True,
                'pid': pid,
                'uptime_seconds': uptime_seconds
            }), 200
        else:
            return jsonify({
                'running': False,
                'pid': None
            }), 200
            
    except Exception as e:
        log.error(f"Error checking bot status: {e}")
        return jsonify({
            'error': str(e),
            'running': False
        }), 500

@bot_control_bp.route('/api/bot/start', methods=['POST'])
def start_bot():
    """
    Start the trading bot in tmux session
    
    Returns:
        JSON response with success/failure status
    """
    try:
        # Check if already running
        pid = _read_pid_file(BOT_PID_FILE)
        if pid and _is_process_running(pid):
            return jsonify({
                'error': 'Bot already running',
                'pid': pid
            }), 400
        
        # Start bot in tmux session
        cmd = [
            'tmux', 'new-session', '-d', '-s', 'gridbot',
            'cd ~/Projects/WorkingBot && python3 bot/run.py'
        ]
        
        subprocess.run(cmd, check=True)
        
        # Wait briefly for PID file to be created
        import time
        time.sleep(1)
        
        new_pid = _read_pid_file(BOT_PID_FILE)
        
        return jsonify({
            'success': True,
            'message': 'Bot started successfully',
            'pid': new_pid
        }), 200
        
    except subprocess.CalledProcessError as e:
        log.error(f"Failed to start bot: {e}")
        return jsonify({
            'error': 'Failed to start bot',
            'details': str(e)
        }), 500
    except Exception as e:
        log.error(f"Error starting bot: {e}")
        return jsonify({'error': str(e)}), 500

@bot_control_bp.route('/api/bot/stop', methods=['POST'])
def stop_bot():
    """
    Stop the trading bot gracefully
    
    Returns:
        JSON response with success/failure status
    """
    try:
        pid = _read_pid_file(BOT_PID_FILE)
        
        if not pid:
            return jsonify({
                'error': 'Bot not running (no PID file)'
            }), 400
        
        if not _is_process_running(pid):
            # Clean up stale PID file
            BOT_PID_FILE.unlink(missing_ok=True)
            return jsonify({
                'error': 'Bot not running (stale PID)',
                'pid': pid
            }), 400
        
        # Stop process
        if _stop_process_by_pid(pid):
            # Clean up PID file
            BOT_PID_FILE.unlink(missing_ok=True)
            
            return jsonify({
                'success': True,
                'message': 'Bot stopped successfully',
                'pid': pid
            }), 200
        else:
            return jsonify({
                'error': 'Failed to stop bot',
                'pid': pid
            }), 500
            
    except Exception as e:
        log.error(f"Error stopping bot: {e}")
        return jsonify({'error': str(e)}), 500
```
